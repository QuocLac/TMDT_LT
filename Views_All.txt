    <script src="https://cdnjs.cloudflare.com/ajax/libs/axios/0.21.1/axios.min.js"></script>
    <script>
        document.addEventListener("DOMContentLoaded", function() {

            // 1. HIỂN THỊ TOAST TỪ TEMPDATA
            @if (TempData["Success"] != null)
            {
                    <text>window.showToast('@TempData["Success"]', 'success');</text>
            }
            @if (TempData["Error"] != null)
            {
                    <text>window.showToast('@TempData["Error"]', 'error');</text>
            }

            // 2. XỬ LÝ CHUYỂN TAB
            const tabs = document.querySelectorAll('.tab-btn');
            const contents = document.querySelectorAll('.tab-content');
            const initialTab = document.getElementById('initialTab').value || 'profile';

            function activateTab(tabName) {
                tabs.forEach(t => t.classList.remove('active'));
                contents.forEach(c => c.classList.remove('active'));

                const targetBtn = document.querySelector(`.tab-btn[data-target="${tabName}"]`);
                const targetContent = document.getElementById(`tab-${tabName}`);

                if (targetBtn && targetContent) {
                    targetBtn.classList.add('active');
                    targetContent.classList.add('active');
                }
            }

            activateTab(initialTab);

            tabs.forEach(tab => {
                tab.addEventListener('click', () => {
                    activateTab(tab.getAttribute('data-target'));
                });
            });

            // 3. API VIỆT NAM PROVINCES
            const cit = document.getElementById("apiCity");
            const dis = document.getElementById("apiDistrict");
            const ward = document.getElementById("apiWard");

            axios.get('https://provinces.open-api.vn/api/?depth=3')
            .then(function (response) {
                renderCity(response.data);
            });

            function renderCity(data) {
                for (const x of data) {
                    cit.options[cit.options.length] = new Option(x.name, x.name);
                }
                cit.onchange = function () {
                    dis.length = 1; ward.length = 1;
                    if(this.value !== "") {
                        const result = data.filter(n => n.name === this.value);
                        for (const k of result[0].districts) {
                            dis.options[dis.options.length] = new Option(k.name, k.name);
                        }
                    }
                };
                dis.onchange = function () {
                    ward.length = 1;
                    const dataCity = data.filter((n) => n.name === cit.value);
                    if (this.value !== "") {
                        const dataWards = dataCity[0].districts.filter(n => n.name === this.value)[0].wards;
                        for (const w of dataWards) {
                            ward.options[ward.options.length] = new Option(w.name, w.name);
                        }
                    }
                };
            }

            // Gộp chuỗi Phường + Tên Đường
            document.getElementById('btnSaveAddress').addEventListener('click', function() {
                const w = document.getElementById('apiWard').value;
                const h = document.getElementById('apiHouse').value;
                document.getElementById('hiddenStreet').value = h + ", " + w;
            });
        });
    </script>
}

@section Scripts {
    <script>
        // 1. TÍNH NĂNG CHỌN TẤT CẢ SẢN PHẨM
        document.getElementById('selectAllCheckbox')?.addEventListener('change', function() {
            const isChecked = this.checked;
            document.querySelectorAll('.item-checkbox').forEach(cb => {
                cb.checked = isChecked;
            });
            recalculateSummary();
        });

        // 2. CẬP NHẬT SỐ LƯỢNG SẢN PHẨM BẰNG AJAX VÀ HIỂN THỊ COMPONENT THÔNG BÁO
        function updateQty(variantId, change) {
            const input = document.getElementById(`qty-${variantId}`);
            let newQty = parseInt(input.value) + change;
            if (newQty < 1) newQty = 1;

            const formData = new FormData();
            formData.append('variantId', variantId);
            formData.append('quantity', newQty);

            fetch('/Cart/UpdateQuantity', {
                method: 'POST',
                body: formData
            })
            .then(response => response.json())
            .then(data => {
                if (data.success) {
                    input.value = data.actualQuantity;
                    document.getElementById(`price-${variantId}`).innerText = data.itemTotalPrice;

                    // Nếu khách bấm quá số lượng tồn kho -> Gọi component thông báo lỗi (Màu đỏ)
                    if (data.limitReached) {
                        if (typeof window.showToast === 'function') {
                            window.showToast(data.message, 'error');
                        }
                    }

                    recalculateSummary();
                } else {
                    // Nếu lỗi từ server -> Gọi component thông báo lỗi
                    if (typeof window.showToast === 'function') {
                        window.showToast(data.message, 'error');
                    }
                }
            })
            .catch(error => {
                console.error('Lỗi kết nối API:', error);
                if (typeof window.showToast === 'function') {
                    window.showToast('Lỗi kết nối đến máy chủ.', 'error');
                }
            });
        }

        // 3. THUẬT TOÁN TÍNH TOÁN LẠI TỔNG TIỀN & VOUCHER DỰA TRÊN CHECKBOX
        function recalculateSummary() {
            const checkedBoxes = document.querySelectorAll('.item-checkbox:checked');
            const selectedIds = Array.from(checkedBoxes).map(cb => parseInt(cb.value));

            document.getElementById('lblCheckoutCount').innerText = selectedIds.length;
            document.getElementById('checkoutSelectedItems').value = selectedIds.join(',');

            fetch('/Cart/GetCartSummary', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(selectedIds)
            })
            .then(res => res.json())
            .then(data => {
                if (data.success) {
                    document.getElementById('summarySubtotal').innerText = data.subtotal;
                    updateSmartVoucherUI(data.vouchers, data.rawSubtotal);
                }
            })
            .catch(err => console.error('Lỗi lấy summary:', err));

            const allBoxes = document.querySelectorAll('.item-checkbox');
            const selectAllCb = document.getElementById('selectAllCheckbox');
            if(selectAllCb && allBoxes.length > 0) {
                selectAllCb.checked = (checkedBoxes.length === allBoxes.length);
            }
        }

        // 4. CẬP NHẬT GIAO DIỆN VOUCHER (ĐỘNG)
        function updateSmartVoucherUI(vouchers, currentSubtotal) {
            const container = document.getElementById('smartVoucherContainer');
            const hintText = document.getElementById('voucherHintText');
            const progressBar = document.getElementById('voucherProgressBar');
            const discountElement = document.getElementById('summaryDiscount');
            const totalElement = document.getElementById('summaryTotal');

            if (!vouchers || vouchers.length === 0 || currentSubtotal === 0) {
                container.style.display = 'none';
                discountElement.innerText = "Chưa tính";
                discountElement.style.color = "";
                discountElement.style.fontWeight = "normal";
                totalElement.innerText = new Intl.NumberFormat('vi-VN').format(currentSubtotal) + " đ";
                return;
            }

            container.style.display = 'block';
            const bestVoucher = vouchers[0];
            let discountAmount = 0;

            if (bestVoucher.isEligible) {
                progressBar.style.width = '100%';
                progressBar.style.background = '#10B981';
                hintText.innerHTML = `Tuyệt vời! Đủ điều kiện sử dụng mã <strong>${bestVoucher.code}</strong>.`;
                discountAmount = bestVoucher.estimatedDiscountAmount;
            } else {
                let percent = (currentSubtotal / bestVoucher.minOrderValue) * 100;
                if (percent > 100) percent = 100;

                progressBar.style.width = percent + '%';
                progressBar.style.background = '#EA580C';

                const formatGap = new Intl.NumberFormat('vi-VN').format(bestVoucher.gapAmount);
                const discountDisplay = bestVoucher.discountType === 1
                    ? new Intl.NumberFormat('vi-VN').format(bestVoucher.discountValue) + " đ"
                    : bestVoucher.discountValue + "%";

                hintText.innerHTML = `Mua thêm <strong>${formatGap} đ</strong> để áp dụng mã <strong>${bestVoucher.code}</strong> (Giảm ${discountDisplay}).`;
            }

            if (discountAmount > 0) {
                discountElement.innerText = "- " + new Intl.NumberFormat('vi-VN').format(discountAmount) + " đ";
                discountElement.style.color = "#DC2626";
                discountElement.style.fontWeight = "700";
            } else {
                discountElement.innerText = "Chưa tính";
                discountElement.style.color = "";
                discountElement.style.fontWeight = "normal";
            }

            let finalTotal = currentSubtotal - discountAmount;
            if (finalTotal < 0) finalTotal = 0;
            totalElement.innerText = new Intl.NumberFormat('vi-VN').format(finalTotal) + " đ";
        }

        // 5. RÀNG BUỘC KHI BẤM NÚT MUA HÀNG VỚI COMPONENT THÔNG BÁO
        document.getElementById('btnCheckout')?.addEventListener('click', function() {
            const selectedItems = document.getElementById('checkoutSelectedItems').value;

                // Thay thế alert() bằng component thông báo Toast

                }
                return;
            }
            document.getElementById('checkoutForm').submit();
        });

        // Chạy tính toán 1 lần khi load trang để reset Subtotal về 0 đ
        document.addEventListener('DOMContentLoaded', recalculateSummary);
    </script>
}

        window.removeFromFavorites = async function (event, variantId) {
            event.preventDefault();
            event.stopPropagation();

            if (!confirm("Xóa sản phẩm này khỏi danh sách yêu thích?")) return;

            try {
                const formData = new FormData();

                // Gọi vào hàm Toggle, nếu truyền mã đang có sẵn nó sẽ tự Xóa

                    body: formData
                });

                const data = await response.json();

                if (data.success) {
                    window.showToast("Đã xóa khỏi danh sách.", "success");

                    // Hiệu ứng xóa mờ dần thẻ HTML
                    const card = document.getElementById(`fav-item-${variantId}`);
                    if (card) {
                        card.style.transition = "all 0.3s ease";
                        card.style.opacity = "0";
                        card.style.transform = "scale(0.9)";

                        setTimeout(() => {
                            card.remove();
                            // Nếu xóa hết sạch sản phẩm, reload trang để hiện màn hình "Trống"
                            if (document.querySelectorAll('.market-product-card').length === 0) {
                                window.location.reload();
                            }
                        }, 300);
                    }
                } else {
                    window.showToast(data.message, "error");
                }
            } catch (e) {
                window.showToast("Lỗi kết nối máy chủ.", "error");
            }
        };

        // 2. TÍNH NĂNG THÊM GIỎ HÀNG KHÔNG RELOAD TRANG

                btnSubmit.innerHTML = '<i class="fas fa-spinner fa-spin"></i> ĐANG XỬ LÝ';

                try {
                    const formData = new FormData(this);

                    const response = await fetch(url, {
                        method: 'POST',
                        body: formData,
                        headers: { 'X-Requested-With': 'XMLHttpRequest' }
                    });

                    if (response.ok || response.redirected) {
                        window.showToast("Đã thêm sản phẩm vào giỏ hàng!", "success");

    <script>
        // Toast
        window.showToast = function(message, type = 'success') {
            const container = document.getElementById('toast-container');
            const toast = document.createElement('div');
            toast.className = `toast-msg ${type}`;
            const icon = type === 'success' ? '<i class="fas fa-check-circle success-icon"></i>' : '<i class="fas fa-exclamation-circle error-icon"></i>';
            toast.innerHTML = `${icon} <span>${message}</span>`;
            container.appendChild(toast);
            setTimeout(() => toast.classList.add('show'), 10);
            setTimeout(() => {
                toast.classList.remove('show');
                setTimeout(() => toast.remove(), 400);
            }, 3000);
        };

        // ================================================================
        // TÌM KIẾM GỢI Ý DROPDOWN (KIỂU SHOPEE) - TOÀN CỤC
        // ================================================================
        document.addEventListener('DOMContentLoaded', function() {
            const searchInput = document.getElementById('global-search-input');
            const suggestionsBox = document.getElementById('global-suggestions-box');
            const searchForm = document.getElementById('global-search-form');
            let debounceTimer;

            if (!searchInput || !suggestionsBox) {
                console.warn('Không tìm thấy input hoặc dropdown gợi ý');
                return;
            }

            // Hàm fetch gợi ý từ API
            function fetchSuggestions(query) {
                const url = `/Store/GetSearchSuggestions?q=${encodeURIComponent(query)}`;
                fetch(url)
                    .then(res => {
                        if (!res.ok) throw new Error('HTTP ' + res.status);
                        return res.json();
                    })
                    .then(data => {
                        let html = '';
                        if (data.phrases && data.phrases.length > 0) {
                            data.phrases.forEach(phrase => {
                                const safePhrase = phrase.replace(/'/g, "\\'").replace(/"/g, '&quot;');
                                html += `
                                    <div class="shopee-suggestion-row" onclick="window.applySuggestion('${safePhrase}')">
                                        <i class="fas fa-search"></i>
                                        <span>${phrase}</span>
                                    </div>`;
                            });
                            suggestionsBox.innerHTML = html;
                            suggestionsBox.style.display = 'block';
                        } else {
                            suggestionsBox.style.display = 'none';
                        }
                    })
                    .catch(err => {
                        console.error('Lỗi fetch gợi ý:', err);
                        suggestionsBox.style.display = 'none';
                    });
            }

            // Hàm áp dụng gợi ý (global)
            window.applySuggestion = function(phrase) {
                if (!searchInput) return;
                searchInput.value = phrase;
                suggestionsBox.style.display = 'none';
                if (searchForm) {
                    searchForm.submit();
                } else {
                    const submitBtn = document.querySelector('.search-submit-btn');
                    if (submitBtn) submitBtn.click();
                }
            };

            // Sự kiện focus: hiển thị gợi ý
            searchInput.addEventListener('focus', function() {
                const val = this.value.trim();
                if (val !== '') {
                    fetchSuggestions(val);
                } else {
                    fetchSuggestions(''); // gợi ý hot
                }
            });

            // Sự kiện input: debounce 300ms
            searchInput.addEventListener('input', function() {
                clearTimeout(debounceTimer);
                const val = this.value.trim();
                debounceTimer = setTimeout(() => {
                    fetchSuggestions(val);
                }, 300);
            });

            // Ẩn dropdown khi click ra ngoài
            document.addEventListener('click', function(e) {
                if (!searchInput.contains(e.target) && !suggestionsBox.contains(e.target)) {
                    suggestionsBox.style.display = 'none';
                }
            });

            // Phím ESC để đóng dropdown
            searchInput.addEventListener('keydown', function(e) {
                if (e.key === 'Escape') {
                    suggestionsBox.style.display = 'none';
                }
            });

            // Khi submit form (Enter), đóng dropdown
            if (searchForm) {
                searchForm.addEventListener('submit', function() {
                    suggestionsBox.style.display = 'none';
                });
            }

            // Tự động mở dropdown khi trang load với từ khóa có sẵn (trên trang kết quả)
            if (searchInput.value.trim() !== '') {
                fetchSuggestions(searchInput.value.trim());
            }
        });
    </script>

            function fetchShopeeSuggestions(queryVal) {
                const url = `/Store/GetSearchSuggestions?q=${encodeURIComponent(queryVal)}`;
                fetch(url)
                    .then(res => {
                        if (!res.ok) throw new Error('HTTP ' + res.status);
                        return res.json();
                    })
                    .then(data => {
                        let html = '';
                        if (data.phrases && data.phrases.length > 0) {
                            data.phrases.forEach(phrase => {
                                const safePhrase = phrase.replace(/'/g, "\\'").replace(/"/g, '&quot;');
                                html += `
                                    <div class="shopee-suggestion-row" onclick="window.applyPdpSuggestion('${safePhrase}')">
                                        <i class="fas fa-search"></i>
                                        <span>${phrase}</span>
                                    </div>`;
                            });
                            suggestionsBox.innerHTML = html;
                            suggestionsBox.style.display = 'block';
                        } else {
                            suggestionsBox.style.display = 'none';
                        }
                    })
                    .catch(err => {
                        console.error("PDP Suggestions Fail:", err);
                        suggestionsBox.style.display = 'none';
                    });

                (qaCurrentSelection.ram !== null || v.ram === "") &&
